-- Logic aimbot
local RunService = game:GetService("RunService")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

RunService.RenderStepped:Connect(function()
    if not _G.Aimbot then return end
    local target = nil
    local shortest = math.huge

    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            local head = plr.Character:FindFirstChild("Head")
            if hrp and head then
                local dist = (hrp.Position - Camera.CFrame.Position).Magnitude
                if dist < shortest then
                    shortest = dist
                    target = plr
                end
            end
        end
    end

    if target and target.Character then
        local part = target.Character:FindFirstChild(_G.AimPart or "Head")
        if part then
            Camera.CFrame = Camera.CFrame:Lerp(
                CFrame.new(Camera.CFrame.Position, part.Position),
                1 / (_G.AimSmooth or 5)
            )
        end
    end
end)

-- Logic ESP
local espGui = Instance.new("ScreenGui")
espGui.Name = "OrangeESP"
espGui.Parent = game.CoreGui
espGui.ResetOnSpawn = false

local function createESP(plr)
    if plr == LocalPlayer then return end
    local box = Instance.new("BillboardGui")
    box.Name = plr.Name
    box.Size = UDim2.new(0, 100, 0, 50)
    box.AlwaysOnTop = true
    box.StudsOffset = Vector3.new(0, 3, 0)
    box.Parent = espGui
    box.Adornee = nil

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.new(1, 0, 0)
    label.TextStrokeTransparency = 0
    label.Font = Enum.Font.GothamBold
    label.TextSize = 14
    label.Text = plr.Name
    label.Parent = box

    return {Box = box, Label = label}
end

local espCache = {}
for _, plr in pairs(Players:GetPlayers()) do
    if plr ~= LocalPlayer then
        espCache[plr] = createESP(plr)
    end
end

Players.PlayerAdded:Connect(function(plr)
    espCache[plr] = createESP(plr)
end)

Players.PlayerRemoving:Connect(function(plr)
    if espCache[plr] then
        espCache[plr].Box:Destroy()
        espCache[plr] = nil
    end
end)

RunService.RenderStepped:Connect(function()
    for plr, esp in pairs(espCache) do
        if _G.ESP and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = plr.Character.HumanoidRootPart
            esp.Box.Adornee = hrp
            esp.Box.Enabled = true
            local dist = math.floor((hrp.Position - Camera.CFrame.Position).Magnitude)
            esp.Label.Text = plr.Name .. " [" .. dist .. "m]"
            esp.Label.TextColor3 = _G.ESPColor or Color3.fromRGB(255, 0, 0)
        else
            esp.Box.Enabled = false
        end
    end
end)

-- TAB 4: TELEPORT
-- ============================================
local TpTab = Window:MakeTab({
    Title = "Teleport",
    Icon = "move"
})

TpTab:AddSection({Name = "Player Teleport"})

local playerList = {}
for _, plr in pairs(Players:GetPlayers()) do
    table.insert(playerList, plr.Name)
end

local selectedPlayer = nil

local playerDropdown = TpTab:AddDropdown({
    Name = "Select Player",
    Desc = "Chọn người chơi để teleport",
    Options = playerList,
    Default = "",
    Flag = "TPPlayer",
    Callback = function(v)
        selectedPlayer = v
    end
})

Players.PlayerAdded:Connect(function(plr)
    playerDropdown:Add(plr.Name)
end)

TpTab:AddButton({
    Name = "Teleport To Player",
    Desc = "Bay tới người chơi đã chọn",
    Callback = function()
        if not selectedPlayer then
            return
        end
        local target = Players:FindFirstChild(selectedPlayer)
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            local myChar = LocalPlayer.Character
            if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                myChar.HumanoidRootPart.CFrame = target.Character.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
            end
        end
    end
})

TpTab:AddSection({Name = "Coordinate Teleport"})

local coordX, coordY, coordZ = 0, 0, 0

TpTab:AddTextBox({
    Name = "X",
    Desc = "Tọa độ X",
    Default = "0",
    PlaceholderText = "X",
    Callback = function(v) coordX = tonumber(v) or 0 end
})

TpTab:AddTextBox({
    Name = "Y",
    Desc = "Tọa độ Y",
    Default = "0",
    PlaceholderText = "Y",
    Callback = function(v) coordY = tonumber(v) or 0 end
})

TpTab:AddTextBox({
    Name = "Z",
    Desc = "Tọa độ Z",
    Default = "0",
    PlaceholderText = "Z",
    Callback = function(v) coordZ = tonumber(v) or 0 end
})

TpTab:AddButton({
    Name = "Teleport To Coordinates",
    Desc = "Bay tới tọa độ đã nhập",
    Callback = function()
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = CFrame.new(coordX, coordY, coordZ)
        end
    end
